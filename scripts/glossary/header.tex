% xelatex header for the generated Dhamma glossary PDF.

\usepackage{fontspec}
\usepackage{xeCJK}

% Generate /ActualText markers so PDF text extraction returns the source
% characters regardless of glyph-to-CMap mapping. Locally on macOS the main
% font resolves to Times New Roman, which has proper ToUnicode entries for
% U+2194 (↔) and the ff / fi / fl ligature codepoints. On CI the main-font
% fallback chain (Times New Roman ✗ → FreeSerif ✗ → TeX Gyre Termes ✗) lands
% on Latin Modern Roman, whose ToUnicode coverage for those characters is
% incomplete — the deployed PDF extracts the arrows as U+FFFD replacement
% characters and the ligatures as their FB00-block codepoints. ActualText
% bypasses both issues.
\XeTeXgenerateactualtext=1

\IfFontExistsTF{Times New Roman}{
  \setmainfont{Times New Roman}
}{
  \IfFontExistsTF{FreeSerif}{
    \setmainfont{FreeSerif}
  }{
    \IfFontExistsTF{TeX Gyre Termes}{
      \setmainfont{TeX Gyre Termes}
    }{
      \setmainfont{Latin Modern Roman}
    }
  }
}

\setCJKmainfont{Noto Serif CJK SC}
\setCJKsansfont{Noto Sans CJK SC}
\setCJKmonofont{Noto Sans Mono CJK SC}

% Keep xeCJK from applying CJK punctuation spacing to English prose.
\xeCJKDeclareCharClass{HalfLeft}{`“}
\xeCJKDeclareCharClass{HalfRight}{`”}
\xeCJKDeclareCharClass{Default}{`—, `–, `‘, `’}

% Hebrew script fallback for Part XV Kabbalah entries (Times New Roman /
% FreeSerif have no reliable Hebrew vowel-pointing coverage).
% Fallback chain: scholarly fonts first (Noto/SBL/Ezra ship with most Linux
% TeX distros and via Homebrew), then macOS built-ins (Arial Hebrew Scholar
% has full Biblical-Hebrew vowel-pointing; New Peninim MT is a serif option;
% Arial Hebrew is the basic macOS Hebrew font).
\IfFontExistsTF{Noto Sans Hebrew}{
  \newfontfamily\hebrewfont{Noto Sans Hebrew}
}{
  \IfFontExistsTF{SBL Hebrew}{
    \newfontfamily\hebrewfont{SBL Hebrew}
  }{
    \IfFontExistsTF{Ezra SIL}{
      \newfontfamily\hebrewfont{Ezra SIL}
    }{
      \IfFontExistsTF{Arial Hebrew Scholar}{
        \newfontfamily\hebrewfont{Arial Hebrew Scholar}
      }{
        \IfFontExistsTF{New Peninim MT}{
          \newfontfamily\hebrewfont{New Peninim MT}
        }{
          \IfFontExistsTF{Arial Hebrew}{
            \newfontfamily\hebrewfont{Arial Hebrew}
          }{
            \newfontfamily\hebrewfont{Times New Roman}
          }
        }
      }
    }
  }
}
% Greek script fallback for Part XV Plotinian / Patristic entries.
% Times New Roman locally has Greek; CI Linux without TNR may fall through
% to a font without polytonic-Greek coverage. Fallback chain mirrors Hebrew:
% scholarly fonts first (Noto Serif ships with fonts-noto-core, already
% installed for the Hebrew fix), then macOS built-ins.
\IfFontExistsTF{Noto Serif}{
  \newfontfamily\greekfont{Noto Serif}
}{
  \IfFontExistsTF{GFS Didot}{
    \newfontfamily\greekfont{GFS Didot}
  }{
    \IfFontExistsTF{Linux Libertine O}{
      \newfontfamily\greekfont{Linux Libertine O}
    }{
      \newfontfamily\greekfont{Times New Roman}
    }
  }
}

% Track Hebrew + Greek + Latin blocks so the script→Latin exit transitions fire
% (without a tracked target block, the exit doesn't fire and subsequent
% Latin text gets typeset in the wrong font). CJK blocks are left
% untracked so xeCJK's existing automatic CJK font switching still works.
%
% GreekExtended (U+1F00–U+1FFF) is tracked separately from GreekAndCoptic
% (U+0370–U+03FF) because precomposed polytonic characters (vowels with
% breathing marks: ἀ, ὑ, ἕ, etc.) live in the Extended block. Without
% GreekExtended tracked, words that *start* with such a character don't
% trigger the Latin→Greek transition and render in the main Latin font
% (which lacks polytonic-Greek coverage in CI).
\usepackage[Hebrew, GreekAndCoptic, GreekExtended, Latin]{ucharclasses}
\setTransitionsFor{Hebrew}{\hebrewfont}{\normalfont}
\setTransitionsFor{GreekAndCoptic}{\greekfont}{\normalfont}
\setTransitionsFor{GreekExtended}{\greekfont}{\normalfont}

% Suppress font-switching at the GreekAndCoptic↔GreekExtended boundary.
% Both blocks share \greekfont, but \setTransitionsFor writes both an
% "enter" and a "leave" token to every cross-pair register, and whichever
% \setTransitionsFor call runs second wins. As written above, the GE→GAC
% pair register ends up holding \normalfont (the "leave GE" token from
% line 93), which strips the Greek font from any GAC character that
% follows a GE character within a word (the σ in νοῦς, the ν in τὸ ἕν,
% the σία in οὐσία). On CI the main font lacks polytonic Greek, so those
% trailing characters render as missing glyphs. Nulling both directions
% of the pair keeps \greekfont active across the boundary.
\XeTeXinterchartoks \GreekAndCopticClass \GreekExtendedClass = {}
\XeTeXinterchartoks \GreekExtendedClass \GreekAndCopticClass = {}
